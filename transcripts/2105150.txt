import os
import jsonlines
import easyocr
from jiwer import cer

# 1. Define paths to your grading kit data
LABELS_PATH = "../grading_kit/labels.jsonl"
PAGES_DIR = "../grading_kit/heldout_pages"

# Lists to store ground truth and OCR predictions
ground_truths = []
predictions = []

print("Starting OCR evaluation using EasyOCR (bn+en) on 9 held-out pages...\n")

# 2. Initialize EasyOCR (it will download the bn and en model weights the first time you run this)
# Setting gpu=False ensures it runs on CPU if you don't have a configured GPU
reader = easyocr.Reader(['bn', 'en'], gpu=False)

# 3. Open and read the labels.jsonl file
try:
    with jsonlines.open(LABELS_PATH) as f:
        for obj in f:
            file_name = obj.get("page_id")
            true_text = obj.get("text", "").strip()
            
            # Construct the full path to the image
            image_path = os.path.join(PAGES_DIR, file_name)
            
            # Skip if file doesn't exist
            if not os.path.exists(image_path):
                print(f"Warning: Image {file_name} not found in {PAGES_DIR}")
                continue
                
            # 4. Run EasyOCR
            # detail=0 returns just a list of extracted text strings (ignoring bounding box coordinates for now)
            result = reader.readtext(image_path, detail=0)
            
            # Join the strings with a space (as Abdullah noted, EasyOCR doesn't do newlines well)
            pred_text = " ".join(result).strip()
            
            # 5. Store the results
            ground_truths.append(true_text)
            predictions.append(pred_text)
            
            print(f"✅ Processed {file_name}")
            
except FileNotFoundError:
    print(f"❌ Error: Could not find {LABELS_PATH}")

# 6. Calculate overall Character Error Rate (CER)
if ground_truths and predictions:
    overall_cer = cer(ground_truths, predictions)
    
    print("\n" + "="*40)
    print("🎯 OCR EVALUATION RESULTS (EasyOCR)")
    print("="*40)
    print(f"Total Pages Evaluated: {len(ground_truths)}")
    print(f"Overall Character Error Rate (CER): {overall_cer:.4f}")
    
    # Calculate Accuracy as (1 - CER)
    accuracy = max(0, 1 - overall_cer)
    print(f"Estimated Character Accuracy: {accuracy:.2%}")
else:
    print("No data was evaluated. Check your labels.jsonl file.")

import faiss
import numpy as np
import re
from sentence_transformers import SentenceTransformer

print("1. Loading Embedding Model (multilingual-e5-large)...")
# Note: e5-large requires "query: " and "passage: " prefixes
embedder = SentenceTransformer('intfloat/multilingual-e5-large')

print("2. Chunking the OCR text (Sentence-boundary semantic chunking)...")
chunks = []
chunk_metadata = []

# Using the `predictions` list from your previous EasyOCR cell
for i, text in enumerate(predictions):
    # Split by Bangla danda (।) or Latin period (.)
    sentences = re.split(r'[।.]', text)
    
    for sentence in sentences:
        clean_sentence = sentence.strip()
        if len(clean_sentence) > 20: # Keep meaningful chunks only
            chunks.append(clean_sentence)
            chunk_metadata.append({"page_index": i, "source_text": clean_sentence})

print(f"Created {len(chunks)} chunks from the 9 pages.")

print("3. Generating Embeddings...")
# E5 models expect the "passage: " prefix for indexing
passage_chunks = [f"passage: {c}" for c in chunks]
embeddings = embedder.encode(passage_chunks, normalize_embeddings=True)

print("4. Building FAISS IndexFlatIP...")
dimension = embeddings.shape[1]
# IndexFlatIP uses Inner Product (which equals Cosine Similarity since we normalized)
index = faiss.IndexFlatIP(dimension)
index.add(embeddings)

print("\n" + "="*40)
print("📊 PHASE 2: INDEX STATISTICS")
print("="*40)
print(f"Total Chunks Indexed: {index.ntotal}")
print(f"Embedding Dimension: {dimension}")
print(f"Index Type: FAISS IndexFlatIP")
print(f"Corpus Coverage: 9 pages / {sum([len(c.split()) for c in chunks])} words")

print("\n" + "="*40)
print("🔍 PHASE 3: ONE RETRIEVAL EXAMPLE")
print("="*40)

# Let's search for something related to Biology/Cells or Physics (adjust if your 9 pages are different)
test_query = "জীববিজ্ঞানের শাখা কয়টি ও কী কী?" # "What are the branches of biology?"
print(f"Query: '{test_query}'")

# E5 expects the "query: " prefix for searching
query_embedding = embedder.encode([f"query: {test_query}"], normalize_embeddings=True)

# Retrieve top 1 result
k = 1
distances, indices = index.search(query_embedding, k)

top_idx = indices[0][0]
score = distances[0][0]
retrieved_text = chunk_metadata[top_idx]["source_text"]

print(f"\nTop Match Score (Cosine Similarity): {score:.4f}")
print(f"Retrieved Chunk:\n{retrieved_text}")
